/* ── API 호출 (환경변수에서 키 읽기) ── */
async function callClaude(messages, system) {
  const apiKey = process.env.REACT_APP_ANTHROPIC_API_KEY;
  if (!apiKey) return "⚠️ API 키가 설정되지 않았습니다. .env 파일에 REACT_APP_ANTHROPIC_API_KEY를 입력해주세요.";
  const r = await fetch("https://api.anthropic.com/v1/messages", {
    method:"POST",
    headers:{ "Content-Type":"application/json", "x-api-key": apiKey, "anthropic-version":"2023-06-01" },
    body: JSON.stringify({ model:"claude-sonnet-4-20250514", max_tokens:1000, system, messages })
  });
  const d = await r.json();
  return d.content?.map(b => b.text||"").join("") || "오류가 발생했습니다.";
}

    </>
  );
}

function LetterTab() {

  const [input, setInput] = useState("");
  const [result, setResult] = useState("");
  const [loading, setLoading] = useState(false);
  const [copied, setCopied] = useState(false);
  const p = P.letter;
  async function process() {
    if (!input.trim()) return;
    setLoading(true); setResult("");
    const sys = mode==="d2s"
      ? `${region} 사투리 편지를 따뜻하고 자연스러운 표준어 편지로 변환. 원래 감정을 살려주세요.`

    const res = await callClaude([{ role:"user", content:input }], sys);
    setResult(res); setLoading(false);

  function copy() { navigator.clipboard?.writeText(result); setCopied(true); setTimeout(() => setCopied(false), 1800); }

"not-allowed" : "pointer", display:"flex", alignItems:"center", justifyContent:"center", gap:8 }}>

      <div style={{ position:"fixed", bottom:0, left:"50%", transform:"translateX(-50%)", width:"100%", maxWidth:520, background:"rgba(10,10,10,0.95)", backdropFilter:"blur(20px)", borderTop:"1px solid #ffffff14", padding:"10px 0 20px" }}>
        <div style={{ display:"grid", gridTemplateColumns:"repeat(5, 1fr)" }}>
          {TABS.map(t => {
            const active = tab===t.id;
            const tp = P[t.id];
            return (
              <button key={t.id} onClick={() => setTab(t.id)} style={{ display:"flex", flexDirection:"column", alignItems:"center", gap:4, background:"transparent", border:"none", cursor:"pointer", padding:"4px 0" }}>
                <div style={{ width:44, height:32, borderRadius:16, background:active ? `${tp.accent}28` : "transparent", display:"flex", alignItems:"center", justifyContent:"center", fontSize:18, transition:"background .2s" }}>
                  {t.emoji}
                </div>
                <span style={{ fontSize:10, color:active ? tp.accent : "#555", fontWeight:active ? 600 : 400 }}>{t.label}</span>
              </button>
            );
          })}
        </div>
      </div>
    </div>
  );
}
